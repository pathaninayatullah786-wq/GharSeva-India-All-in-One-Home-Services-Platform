def money(amount):
    return f"₹{amount:,.0f}"


def show_card(title, image, description, price=None):
    with st.container(border=True):
        st.image(image, use_container_width=True)
        st.markdown(f"### {title}")
        st.write(description)
        if price is not None:
            st.markdown(
                f'<div class="price">{money(price)}</div>',
                unsafe_allow_html=True
            )

# -------------------- TOOLS SHOP --------------------

elif page == "🛒 Tools Shop":
    st.title("🛒 Home-Service Tools Shop")
    st.write("Choose a worker category to see the related tools.")

    job = st.selectbox(
        "Tools for",
        ["All Workers"] + [w["job"] for w in WORKERS]
    )

    all_tools = []
    for worker in WORKERS:
        if job != "All Workers" and worker["job"] != job:
            continue
        for tool in worker["tools"]:
            all_tools.append({
                **tool,
                "job": worker["job"]
            })

    cols = st.columns(3)
    for i, tool in enumerate(all_tools):
        with cols[i % 3]:
            show_card(
                tool["name"],
                photo(tool["image"], 600),
                f'{tool["job"] if "job" in tool else job} · {tool["description"]}',
                tool["price"]
            )
            qty = st.number_input(
                "Quantity",
                min_value=1,
                max_value=10,
                value=1,
                key=f"shop_qty_{i}"
            )
            if st.button("Add to Cart", key=f"shop_add_{i}"):
                st.session_state.cart.append({
                    "name": tool["name"],
                    "price": tool["price"],
                    "quantity": qty,
                    "job": tool["job"]
                })
                st.success("Tool added!")

    st.divider()
    st.header("🛍️ Your Cart")

    if st.session_state.cart:
        total = 0
        for i, item in enumerate(st.session_state.cart):
            subtotal = item["price"] * item["quantity"]
            total += subtotal
            st.write(
                f'{i + 1}. {item["name"]} × {item["quantity"]} '
                f'— {money(subtotal)}'
            )

        st.markdown(f"## Cart total: {money(total)}")

        with st.form("tool_order_form"):
            customer = st.text_input("Customer name")
            phone = st.text_input("10-digit mobile number")
            address = st.text_area("Delivery address")
            payment = st.radio(
                "Payment method",
                ["Cash on Delivery", "Online Payment (Demo)"]
            )
            submitted = st.form_submit_button("Confirm Tool Order")

        if submitted:
            if not customer.strip() or not address.strip():
                st.error("Enter customer name and delivery address.")
            elif not phone.isdigit() or len(phone) != 10:
                st.error("Enter a valid 10-digit mobile number.")
            else:
                order = {
                    "reference": uuid.uuid4().hex[:8].upper(),
                    "customer": customer,
                    "phone": phone,
                    "address": address,
                    "items": list(st.session_state.cart),
                    "total": total,
                    "payment": payment
                }
                st.session_state.orders.append(order)
                st.session_state.last_result = {
                    "type": "Tool order",
                    "reference": order["reference"],
                    "amount": total,
                    "payment": payment
                }
                st.session_state.cart = []
                st.success("Order saved successfully!")
                st.balloons()
    else:
        st.info("Your cart is empty. Add a tool to get started.")
